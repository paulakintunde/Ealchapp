```typescript
/** Every kind the report accounts for, in print order. `speakPath` was missing
 *  from publish-content.ts's counts entirely — a report that omits a kind the
 *  guard now protects is the same blind spot in a different place. */
export const REPORT_KINDS = [
  'domains', 'themes', 'units', 'lessons', 'items',
  'scenarios', 'playlists', 'speakPath', 'examTasks', 'examPapers',
] as const;

export function formatDiffReport(args: {
  version: number;
  previous?: { version: number; counts?: Record<string, number> | null } | null;
  counts: Record<string, number>;
  noop: boolean;
  compareNote?: string;
  checksum: string;
  contentDigest: string;
  snapshotBytes: number;
  previousSnapshotBytes?: number | null;
  dryRun: boolean;
  generatedAt: string;
}): string;
```

Body, producing markdown in this exact structure:
```
# Publish report

Generated: {generatedAt}
Mode: {dryRun ? 'dry run (nothing is written or uploaded)' : 'publish'}
Status: {one of the three status lines below}

{compareNote ? '> Could not compare content against v{previous.version}: {compareNote}\n> Publishing anyway. The counts below are still real; a no-op cannot be ruled out.\n' : ''}
## What changed

| kind | previous | now | delta |
|---|---|---|---|
| domains | {p} | {n} | {+d} |
... one row per REPORT_KINDS entry ...

## Size

| | previous | now | delta |
|---|---|---|---|
| snapshot bytes | {prevBytes ?? '—'} | {snapshotBytes} | {signed delta, or '—'} |

checksum: {checksum}
content digest: {contentDigest}
```

Status line rules:
- no `previous` → `v{version} is the FIRST snapshot.`
- `noop` → `IDENTICAL to v{previous.version} — not one byte of content differs.`
- otherwise → `v{previous.version} → v{version}`

Delta formatting reuses the existing convention from `publish-content.ts:1112-1115` verbatim: `${delta >= 0 ? '+' : ''}${delta}`. Previous count for a kind is `Number(previous?.counts?.[k]) || 0`; when there is no `previous`, print `—` in the previous column and `+{n}` in the delta column. Missing keys in `counts` print `0`, never `undefined` or `NaN`.

Constraints: no `process.env`, no `node:fs`, no `console.*`, no `Date.now()` inside the function — `generatedAt` is passed in so the output is deterministic and testable. Return value ends with exactly one `\n`.

Then append tests to `ealch-admin/scripts/drift-guard.logic.test.ts` covering the eight behaviours above — same conventions as the existing file (`test()` per behaviour, `node:assert` `ok`/`strictEqual`, no `describe`). Include one test named `it reports every kind, including the speakPath the old counts object forgot` that asserts all ten `REPORT_KINDS` labels appear in the output, and one named `it still reports when nothing changed` asserting the no-op branch produces the full counts table (that is D-02 in a test).
