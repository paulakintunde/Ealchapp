```typescript
import type { ExamSection, ExamTask } from '../content/schema.ts';
export type SittingPin = { readonly section: ExamSection; readonly tasks: ReadonlyMap<string, ExamTask> };
/** Monotonic: the first non-null section is frozen; a task id in that section's

 *  Returns `prev` unchanged (same reference) when nothing new resolved. */
export function pinSitting(prev: SittingPin | null, liveSection: ExamSection | null, liveTasks: readonly ExamTask[]): SittingPin | null

```
