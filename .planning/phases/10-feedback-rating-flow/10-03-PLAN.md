    ```sql
    create table if not exists public.problem_report_usage (
      subject_key text    not null,
      day         date    not null,
      count       integer not null default 0,
      primary key (subject_key, day)
    );
    alter table public.problem_report_usage enable row level security;
    revoke all on table public.problem_report_usage from anon, authenticated;

    create or replace function public.problem_report_bump(p_subject text, p_ip text)
    returns table (subject_count integer, global_count integer, ip_count integer)
    language plpgsql
    set search_path = ''
    as $$
    begin
      insert into public.problem_report_usage (subject_key, day, count) values (p_subject, current_date, 1)
      on conflict (subject_key, day) do update set count = problem_report_usage.count + 1
      returning problem_report_usage.count into subject_count;
      insert into public.problem_report_usage (subject_key, day, count) values ('g:all', current_date, 1)
      on conflict (subject_key, day) do update set count = problem_report_usage.count + 1
      returning problem_report_usage.count into global_count;
      if p_ip is not null then
        insert into public.problem_report_usage (subject_key, day, count) values (p_ip, current_date, 1)
        on conflict (subject_key, day) do update set count = problem_report_usage.count + 1
        returning problem_report_usage.count into ip_count;
      end if;
      return next;
    end;
    $$;
    revoke all on function public.problem_report_bump(text, text) from public;
    revoke all on function public.problem_report_bump(text, text) from anon;
    revoke all on function public.problem_report_bump(text, text) from authenticated;

    create or replace function public.problem_reports_purge()
    returns integer
    language plpgsql
    set search_path = ''
    as $$
    declare n integer;
    begin
      delete from public.problem_reports where created_at < now() - interval '12 months';
      get diagnostics n = row_count;
      delete from public.problem_report_usage where day < current_date - 7;
      return n;
    end;
    $$;
    revoke all on function public.problem_reports_purge() from public;
    revoke all on function public.problem_reports_purge() from anon;
    revoke all on function public.problem_reports_purge() from authenticated;

    -- D-20 retention. pg_cron is optional on this project (verified live in plan 10-09).
    -- When present this schedules the purge daily; when absent the edge function
    -- runs the same purge at most once a day per instance.
    do $$
    begin
      if exists (select 1 from pg_extension where extname = 'pg_cron') then
        perform cron.schedule('problem-reports-retention', '17 3 * * *', 'select public.problem_reports_purge()');
      end if;
    end $$;
    ```
