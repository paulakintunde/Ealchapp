        insert into public.problem_report_usage (subject_key, day) values ('g:all', current_date) on conflict do nothing;
        select count into s from public.problem_report_usage where subject_key = p_subject and day = current_date for update;
        if p_ip is not null then select count into i from public.problem_report_usage where subject_key = p_ip and day = current_date for update; end if;
