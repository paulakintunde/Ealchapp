/* ══════════════════════════════════════════════════════════════════════════
 *  §G. THE AUTHORED ROWS
 *
 *  Respell house style: hyphenated syllables, final stressed syllable in caps,
 *  ü for French u, nasals with ⁿ. The three imparfait sounds:
 *
 *    -ais -ait -aient   [ɛ]    EH      four of six cells, one sound
 *    -ions              [jɔ̃]   YOHⁿ
 *    -iez               [je]   YAY
 *
 *  THE DOUBLED i IS A SPELLING POINT, NOT A SOUND POINT. voyions, croyions and
 *  payions are [jjɔ̃] in the dictionary and [jɔ̃] in most mouths, so the
 *  respelling is the same as the present (vwah-YOHⁿ) and the IPA keeps the
 *  dictionary [jj]. étudiions is different: [di.jɔ̃] against étudions' [djɔ̃]
 *  is a real extra syllable, and the respelling shows it (dee-YOHⁿ).
 *
 *  NO U+203F anywhere. Liaison consonants are written onto the next syllable
 *  (noo zay-TYOHⁿ), and IPA runs them on (/nu.ze.tjɔ̃/).
 * ══════════════════════════════════════════════════════════════════════════ */

type Row = Item;
type VC = NonNullable<Item['verbCheck']>;

const FCVR = ['flashcard', 'voiceflash', 'review'] as Item['drills'];
const FCVDR = ['flashcard', 'voiceflash', 'dictation', 'review'] as Item['drills'];

const vc = (infinitive: string, person: VC['person'], number: VC['number']): VC =>
  ({ infinitive, tense: 'imparfait', person, number });

/** A conjugation card. `prompt` is the card front, in the house shape of
 *  fr.a1.cuisine.005 (« manger · présent · nous »). verbCheck asks the publish
 *  gate to run it through the conjugator (open question: imparfait output
 *  shape unverified against verbecc). */
const conj = (
  n: number, fr: string, en: string, ipa: string, respell: string,
  prompt: string, check: VC, tags: string[], drills: Item['drills'] = FCVR,
): Row => ({
  id: E(n), kind: 'phrase', level: LEVEL, theme: THEME, fr, en, ipa, respell,
  tags: ['imparfait', ...tags], drills, version: 1,
  cardType: 'conjugation', prompt, verbCheck: check, grammarPoints: ['imparfait'],
});

const sent = (
  n: number, fr: string, en: string, ipa: string, respell: string,
  tags: string[], drills: Item['drills'] = FCVR,
): Row => ({
  id: E(n), kind: 'sentence', level: LEVEL, theme: THEME, fr, en, ipa, respell,
  tags: ['imparfait', ...tags], drills, version: 1, grammarPoints: ['imparfait'],
});
